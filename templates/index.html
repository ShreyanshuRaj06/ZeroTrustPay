+ def verify_hmac(payload: dict, provided_sig: str) -> bool:
+ canonical = json.dumps({k: payload[k] for k in sorted(payload.keys()) if k != "signature"})
+ expected = hmac.new(HMAC_SECRET, canonical.encode(), hashlib.sha256).hexdigest()
+ return hmac.compare_digest(expected, provided_sig) # Constant-Time
